name: Deploy to GitHub Pages

on:
  push:
    branches: [master]
  workflow_dispatch:

# Allow one concurrent deployment; let a newer push supersede an in-flight one.
concurrency:
  group: pages
  cancel-in-progress: true

permissions:
  contents: read
  pages: write
  id-token: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - run: npm ci
      - name: Typecheck and unit tests
        run: |
          npx tsc -p tsconfig.app.json --noEmit
          npx vitest run
      - run: npx vite build
      - uses: actions/configure-pages@v5
        with:
          # Turn Pages on the first time this runs, so no manual repo-settings step is needed.
          enablement: true
      - uses: actions/upload-pages-artifact@v3
        with:
          path: dist

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
